URLShield is a hybrid URL threat detection engine combining rule-based lexical analysis, behavioural heuristics, and machine learning to identify phishing attempts, malware distribution, and malicious data URIs. Designed around RFC 3986.

class Pipeline:
def run(self, url: str):
url_type = classifier.detect(url)
parsed = parsers[url_type].parse(url)
lex = analyzers.lexical(parsed)
beh = analyzers.behaviour(parsed)
return formatter.build({
"lexical": lex,
"behaviour": beh,
})
